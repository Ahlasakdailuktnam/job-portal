2. Senior Way: Use when()

Laravel មាន method when()

$jobs = Job::with(['company', 'category'])
    ->where('status', 'active')

    ->when($request->keyword, function ($query, $keyword) {
        $query->where('title', 'like', "%{$keyword}%");
    })

    ->when($request->category, function ($query, $category) {
        $query->where('category_id', $category);
    })

    ->when($request->job_type, function ($query, $jobType) {

3. Better: Create Filter Scope

ក្នុង Model

class Job extends Model
{
    public function scopeFilter($query, array $filters)
    {
        $query->when(
            $filters['keyword'] ?? false,
            fn($q, $keyword) =>
                $q->where('title', 'like', "%{$keyword}%")
        );

        $query->when(
            $filters['category'] ?? false,
            fn($q, $category) =>
                $q->where('category_id', $category)
        );

        $query->when(
            $filters['job_type'] ?? false,
            fn($q, $jobType) =>
                $q->where('job_type', $jobType)
        );
    }
}

Controller

public function index(Request $request)
{
    $jobs = Job::with([
            'company',
            'category'
        ])
        ->where('status', 'active')
        ->filter($request->all())
        ->latest()
        ->paginate(10);

    return response()->json([
        'success' => true,
        'data' => $jobs
    ]);
}

4. Most Professional: Dedicated Filter Class

Folder

app/
 └── Filters/
      └── JobFilter.php

JobFilter.php

class JobFilter
{
    public function apply($query, Request $request)
    {
        return $query

            ->when(
                $request->keyword,
                fn($q) =>
                    $q->where(
                        'title',
                        'like',
                        "%{$request->keyword}%"
                    )
            )

            ->when(
                $request->category,
                fn($q) =>
                    $q->where(
                        'category_id',
                        $request->category
                    )
            )
            ->when(
                $request->job_type,
                fn($q) =>
                    $q->where(
                        'job_type',
                        $request->job_type
                    )
            );
    }
}

Controller

public function index(
    Request $request,
    JobFilter $filter
)
{
    $jobs = $filter
        ->apply(
            Job::with([
                'company',
                'category'
            ])
            ->where('status', 'active'),
            $request
        )
        ->latest()
        ->paginate(10);
    return response()->json([
        'success' => true,
        'data' => $jobs
    ]);
}
